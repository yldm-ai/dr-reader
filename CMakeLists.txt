cmake_minimum_required(VERSION 3.24)
project(dr_reader VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
include(FetchContent)
include(CTest)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive- /EHsc)
endif()
FetchContent_Declare(pugixml URL https://github.com/zeux/pugixml/archive/refs/tags/v1.15.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(miniz URL https://github.com/richgel999/miniz/archive/refs/tags/3.0.2.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(pugixml json miniz)
add_subdirectory(engine)
if(WIN32)
  add_subdirectory(ui)
  add_subdirectory(app)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
